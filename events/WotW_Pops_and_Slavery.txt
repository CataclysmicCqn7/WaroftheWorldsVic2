# War of the Worlds - Martian crews and slavery
# Victoria 2 can't check province flags in event triggers and can't change one
# pop's culture, so province modifiers mark state and the crew is "the soldier
# pop of the pit": every other pop in Martian land becomes slaves.
#
# (The crew is created by the arrival chain in WotW_Arrival.txt)
# 777010  Harvest Pens         all non-crew pops in Martian provinces -> slaves, slave state
# 777020  Freed from the Pens  liberated province: slaves -> farmers, modifiers removed

#Harvest Pens
province_event = {
	id = 777010
	title = "EVTNAME777010"
	desc = "EVTDESC777010"
	picture = "Slaves"

	trigger = {
		owner = { tag = MAR }
		any_pop = {
			NOT = { type = slaves }
			NOT = { type = soldiers }
		}
		# A pit that hasn't opened yet waits for its crew (event 777112)
		NOT = { has_province_modifier = martian_landing_site }
	}

	mean_time_to_happen = { days = 3 }

	option = {
		name = "EVTOPTA777010"
		add_province_modifier = { name = martian_harvest_pens duration = -1 }
		any_pop = {
			limit = {
				NOT = { type = slaves }
				NOT = { type = soldiers }
			}
			pop_type = slaves
		}
		state_scope = { is_slave = yes }
	}
}

#Freed from the Pens
province_event = {
	id = 777020
	title = "EVTNAME777020"
	desc = "EVTDESC777020"
	picture = "Slaves"

	trigger = {
		has_province_modifier = martian_harvest_pens
		NOT = { owner = { tag = MAR } }
	}

	mean_time_to_happen = { days = 3 }

	option = {
		name = "EVTOPTA777020"
		remove_province_modifier = martian_harvest_pens
		remove_province_modifier = martian_pit
		any_pop = {
			limit = { type = slaves }
			pop_type = farmers
			militancy = 3
		}
		state_scope = { is_slave = no }
	}
}

#The Feeding: Martian stipend about every 3 months, more for every province with Harvest Pens; 2% of the slaves are culled
country_event = {
	id = 777330
	title = "EVTNAME777330"
	desc = "EVTDESC777330"
	picture = "Slaves"

	trigger = {
		tag = MAR
	}

	mean_time_to_happen = { days = 90 }

	option = {
		name = "EVTOPTA777330"
		treasury = 4500
		any_country = {
			limit = { NOT = { tag = MAR } }
			relation = { who = MAR value = -200 }
		}
		any_country = {
			limit = { alliance_with = MAR }
			leave_alliance = MAR
		}
		any_owned = {
			limit = { has_province_modifier = martian_harvest_pens }
			owner = { treasury = 300 }
			any_pop = {
				limit = { type = slaves }
				reduce_pop = 0.98
			}
		}
	}
}

#War check every 7 days (scheduled events fire on the exact day; MTTH events are only checked about monthly)
country_event = {
	id = 777322
	title = "EVTNAME777322"
	desc = "EVTDESC777322"
	picture = "Artillery"

	is_triggered_only = yes

	option = {
		name = "EVTOPTA777322"
		country_event = { id = 777325 days = 7 }
		political_reform = yes_slavery
		any_neighbor_country = {
			limit = { NOT = { war_with = MAR } }
			country_event = 777321
		}
		any_greater_power = {
			limit = {
				NOT = { tag = MAR }
				NOT = { war_with = MAR }
			}
			country_event = 777321
		}
		country_event = 777326
	}
}
