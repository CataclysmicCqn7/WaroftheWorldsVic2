# War of the Worlds - the arrival (first wave)
#
# NOTE: events fired with country_event = { id = X days = N } IGNORE their trigger
# block. Every check is therefore done inside the effects with
# any_owned = { limit = { ... } }, which the game does honour.
#
# 777100  Flashes Observed on Mars    every civilized country, Oct 1894 (Mars opposition)
# 777105  A Falling Star              owner of the landing site, ~26 months later (Dec 1896)
# 777110  The Cylinder at Oxford owner of 294: MAR is released at peace.
#                                     A: storm the pit now (war)   B: white flag (Martians attack on day 31)
# 777111  We Have Arrived             MAR: country setup
# 777112  The Lid Falls               MAR, day 10: crew emerges, or the cylinder is destroyed
# 777113  The Cylinder Is Destroyed   MAR: humans hold the pit, the landing is crushed
# 777116  The Fighting Machines Rise  MAR, day 30: the first 12 tripods appear, or the pit is destroyed
# 777115  The Deputation              MAR, day 31 (option B only): Martians declare war
#
# Testing: as the owner of Oxford (294), type "event 777110" in the console.

#Flashes Observed on Mars
country_event = {
	id = 777100
	title = "EVTNAME777100"
	desc = "EVTDESC777100"
	picture = "Invention"

	trigger = {
		civilized = yes
		NOT = { tag = MAR }
		NOT = { has_country_flag = mars_flashes_seen }
		OR = {
			year = 1895
			AND = {
				year = 1894
				month = 9
			}
		}
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = "EVTOPTA777100"
		set_country_flag = mars_flashes_seen
		set_global_flag = mars_flashes
		# The landing comes at the next opposition, about 26 months later
		any_owned = {
			limit = { province_id = 294 }
			owner = { country_event = { id = 777105 days = 780 } }
		}
	}
}

#We Have Arrived
country_event = {
	id = 777111
	title = "EVTNAME777111"
	desc = "EVTDESC777111"
	picture = "Riot"

	is_triggered_only = yes

	option = {
		name = "EVTOPTA777111"
		capital = 294
		capital_scope = {
			add_province_modifier = { name = martian_landing_site duration = -1 }
			add_province_modifier = { name = martian_unscrewing duration = 10 }
		}
		primary_culture = martian
		government = martian_overmind
		ruling_party_ideology = reactionary
		civilized = yes
		political_reform = yes_slavery
		upper_house = {
			ideology = reactionary
			value = 1
		}
		activate_technology = martian_fighting_machines
		add_country_modifier = { name = martian_technology duration = -1 }
		# The Black Smoke (gas attack) is the invention black_smoke, which only MAR can
		# discover (activate_technology works for techs only, not inventions).
		treasury = 10000
		any_country = {
			limit = { NOT = { tag = MAR } }
			relation = { who = MAR value = -200 }
		}
	}
}

#The Lid Falls (day 10)
country_event = {
	id = 777112
	title = "EVTNAME777112"
	desc = "EVTDESC777112"
	picture = "Riot"

	is_triggered_only = yes

	option = {
		name = "EVTOPTA777112"
		# The pit has a soldier pop: it becomes the crew
		any_owned = {
			limit = {
				OR = {
					is_capital = yes
					has_province_modifier = martian_landing_site
				}
				NOT = { has_province_modifier = martian_pit }
				controlled_by = MAR
				has_pop_type = soldiers
			}
			remove_province_modifier = martian_landing_site
			remove_province_modifier = martian_unscrewing
			add_province_modifier = { name = martian_pit duration = -1 }
			add_province_modifier = { name = martian_assembling duration = 25 }
			add_province_modifier = { name = martian_harvest_pens duration = -1 }
			# The crew: each regiment needs 3000 soldiers. A typical soldier pop
			# (Oxford 1836: 1500) times 40 gives 60000: exactly 20 tripods (3000 each).
			# (total_pops can't be used to test a single pop's size.)
			any_pop = {
				limit = { type = soldiers }
				reduce_pop = 40
			}
			# Everyone else in the pit goes into the pens
			any_pop = {
				limit = {
					NOT = { type = soldiers }
					NOT = { type = slaves }
				}
				pop_type = slaves
			}
			state_scope = { is_slave = yes }
		}
		# No soldier pop: a small slice of the local workers becomes the crew
		any_owned = {
			limit = {
				OR = {
					is_capital = yes
					has_province_modifier = martian_landing_site
				}
				NOT = { has_province_modifier = martian_pit }
				controlled_by = MAR
				NOT = { has_pop_type = soldiers }
			}
			remove_province_modifier = martian_landing_site
			remove_province_modifier = martian_unscrewing
			add_province_modifier = { name = martian_pit duration = -1 }
			add_province_modifier = { name = martian_assembling duration = 25 }
			add_province_modifier = { name = martian_harvest_pens duration = -1 }
			any_pop = {
				limit = {
					OR = {
						type = farmers
						type = labourers
					}
				}
				pop_type = soldiers
				reduce_pop = 0.6	# e.g. 73000 labourers -> ~44000 crew
			}
			# Everyone else in the pit goes into the pens
			any_pop = {
				limit = {
					NOT = { type = soldiers }
					NOT = { type = slaves }
				}
				pop_type = slaves
			}
			state_scope = { is_slave = yes }
		}
		# Humans hold the pit: the cylinder is destroyed
		any_owned = {
			limit = {
				is_capital = yes
				NOT = { has_province_modifier = martian_pit }
				NOT = { controlled_by = MAR }
			}
			owner = { country_event = 777113 }
		}
		any_owned = {
			limit = {
				has_province_modifier = martian_landing_site
				NOT = { is_capital = yes }
				NOT = { controlled_by = MAR }
			}
			remove_province_modifier = martian_landing_site
			remove_province_modifier = martian_unscrewing
			remove_core = MAR
		}
	}
}
